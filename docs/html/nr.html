NR C library(MMSE-LSA denoiser)— NR/c_impl

概要

AnrDenoiser(V3-2)是 OM-LSA 族降噪核心:Ephraim-Malah 1985 MMSE-LSA 增益 + Cohen&Berdugo 2002 MCRA 噪聲估計 + 2001 Bayesian SPP。頻域 I/O:輸入/輸出都是 Complex[n_freqs] 頻譜,window / forward FFT / inverse FFT / OLA 全由呼叫端負責(串在 AEC 之後時,NR 不另做 FFT,直接吃 AEC 已算好的線性誤差頻譜 SeSpecFrame.e)。純 DSP、無 NN/NPU 成分;全部 state 收在 AnrDenoiser C struct(caller pool 或 heap)。

串在 AEC 之後的整合碼(Audio_ALG 的 pipelines/mono_aec_nr_res/main.c,每支範例 main 自己初始化 NR、自己寫那十行串接)經 pipelines/samples/se_anr_stage.h 的 se_anr_stage_gain() 使用的 seam 是 anr_process_gain() + anr_get_gain():算出每 bin 增益但不套用,並可把外部殘留回音 PSD R² 折進噪聲底(ξ = S²/(N²+R²),R² 先乘 noise_over_subtraction 同一倍率);MCRA 追蹤器只吃乾淨功率,R² 不污染噪聲估計。RES 增益:單聲道是 AEC 自己在呼叫內算的 frame.g_res_ctx;4ch 的 AudioPipeline4Ch 單元不含 RES(audio_pipeline_4ch_fuse() 留下 g_res_ctx = NULL),應用在單元之後自己建立 libaec 的 AecRes,每 hop 用 se_res_stage_gain() 算出 G_res(見下方 C)。兩者整合端都取 g_total = min(G_nr, G_res)。

如何呼叫使用

下面兩段的識別字都對現行 header 編譯檢查過(-Wall -Wextra -Werror)。細節(對齊、錯誤語意、aliasing、reset)見 nr_c_api.html。

A. 直接使用(呼叫端已有頻譜;static pool 路徑)

#include <stdlib.h>
#include "anr.h"

int nr_freq_domain_demo(const Complex *spectra, int n_hops, Complex *out)
{
    AnrConfig cfg = anr_config_for_mode_grid(16000, 256, ANR_MODE_BALANCED);
    if (!anr_validate_config(&cfg)) return -1;

    size_t bytes = anr_get_mem_size(&cfg);              /* 0 = config 被拒 */
    void *pool = NULL;
    if (bytes == 0 || posix_memalign(&pool, 16, bytes) != 0) return -1;   /* 16-byte 對齊 */

    AnrDenoiser *nr = anr_init(pool, bytes, &cfg);      /* NULL = 未對齊 / 太小 / 無效 */
    if (!nr) { free(pool); return -1; }

    const int n_freqs = anr_get_n_freqs(nr);            /* 129 */
    int rc = 0;
    for (int h = 0; h < n_hops && rc == 0; ++h)       /* 每 hop 恰好 n_freqs 個 bin */
        rc = anr_process(nr, spectra + (size_t)h * n_freqs, out + (size_t)h * n_freqs);
    if (rc != 0) anr_reset(nr);     /* 函式庫不會自動 reset,由呼叫端決定 */
    anr_destroy(nr);                /* pool 實例:no-op */
    free(pool);                     /* pool 由呼叫端釋放 */
    return rc;
}

/* 串接鏈只要增益:extra_psd 與 |spectrum|² 同尺度,可為 NULL */
const float *nr_gain_demo(AnrDenoiser *nr, const Complex *spectrum, const float *extra_psd)
{
    if (anr_process_gain(nr, spectrum, extra_psd, NULL) != 0) return NULL;
    return anr_get_gain(nr, NULL);   /* 借用指標,到下次 process/reset/destroy 前有效 */
}

B. 串在 AEC 之後的一個 hop(mono 整合範例的做法;Chain 是應用自己的 struct)

需要的 header:aec.h、anr.h、audio_stft.h、common/se_stage.h、samples/se_anr_stage.h、utility/simd_kernels.h(後三者在 Audio_ALG/pipelines 與 audiolib)。

int chain_hop(Chain *c, const float *mic, const float *ref, float *out)
{
    SeSpecFrame f;
    aec_process_context(c->aec, mic, ref);
    if (se_frame_from_aec(c->aec, &f) != 0) return -1;

    const float *g_res = f.g_res_ctx;                                /* AEC 自己的 RES 增益 */
    const float *g_nr  = se_anr_stage_gain(c->nr, &f, c->extra, 1);  /* r2/32768² 併入噪聲底 */
    if (!g_res || !g_nr) return -1;                                  /* 呼叫端整鏈 reset */

    sk_min_f32(c->total, g_nr, g_res, f.n_freqs);
    sk_capply_gain_f32(c->spec, f.e, c->total, f.n_freqs);
    if (c->enable_cng && f.cn)
        aec_cng_fill(&c->rng, c->spec, f.cn, g_res, g_nr, SE_CNG_NR_GAIN_FLOOR, f.n_freqs);
    (void)ac_istft_push(c->istft, c->spec, out);
    return 0;
}

C. 4ch:單元之後的一個 hop(RES 是應用自己的 AecRes)

順序與 B 不同:先 process → fuse(E、R²、CN,g_res_ctx = NULL),再 RES 增益,才是 NR 增益。NR 端的呼叫(se_anr_stage_gain() → anr_process_gain())與 B 完全相同,R² 併入噪聲底的規則也相同。範例已在暫存複本以 -Wall -Wextra -Werror 編譯通過(只編譯、未連結)。完整初始化見 Audio_ALG 的 pipelines/4ch_aec_bf_nr_res/audio_pipeline_4ch_static.c。

#include "audio_pipeline_4ch.h"   /* Audio_ALG 4ch unit */
#include "aec_res.h"
#include "anr.h"
#include "audio_stft.h"
#include "common/se_stage.h"
#include "samples/se_anr_stage.h"
#include "utility/simd_kernels.h"

typedef struct Chain4 {            /* the application's own struct */
    AudioPipeline4Ch *unit;
    AecRes *res;                   /* created by the application after the unit */
    AnrDenoiser *nr;
    AcIstft *istft;
    float *extra, *total;          /* n_freqs floats each */
    Complex *spec;                 /* n_freqs bins */
    uint32_t rng;                  /* SE_CNG_RNG_SEED */
} Chain4;

int chain_hop_4ch(Chain4 *c, const float *mics, const float *ref, int vad, float *out)
{
    SeSpecFrame f;
    AudioPipeline4ChBeam beam;
    AudioPipeline4ChFrameInfo info;
    if (audio_pipeline_4ch_process_with_activity(c->unit, mics, ref, 1, vad, NULL, &beam, &info)
            != FOUR_AEC_NR_RES_OK) return -1;
    if (audio_pipeline_4ch_fuse(c->unit, &f) != FOUR_AEC_NR_RES_OK) return -1;  /* f.g_res_ctx == NULL */

    const float *g_res = se_res_stage_gain(c->res, &f);                      /* the app's AecRes */
    const float *g_nr  = g_res ? se_anr_stage_gain(c->nr, &f, c->extra, 1) : NULL;
    if (!g_res || !g_nr) return -1;                                          /* reset the whole chain */

    sk_min_f32(c->total, g_nr, g_res, f.n_freqs);
    sk_capply_gain_f32(c->spec, f.e, c->total, f.n_freqs);
    if (f.cn)
        aec_cng_fill(&c->rng, c->spec, f.cn, g_res, g_nr, SE_CNG_NR_GAIN_FLOOR, f.n_freqs);
    (void)ac_istft_push(c->istft, c->spec, out);
    return 0;
}

延遲與 warm-up

  • 演算法延遲 = 0 樣本:頻域增益同 hop 產出;框架延遲(window+FFT / IFFT+OLA,50% overlap 下為 1 hop:16k/256 = 128 樣本 = 8 ms、16k/512 = 16 ms、48k/1024 = 10.667 ms)由呼叫端的 transform 負擔——mono 整合範例的端到端 1 hop 就來自這層(參見 aec.html 延遲表)。16k/256 的脈衝實測:輸出峰值比輸入晚 128 樣本。
  • Warm-up = num_init_frames(≈200 ms,格點重定時:16k/256=25 幀、16k/512=13 幀、48k=19 幀,實測):期間累積噪聲底、增益=1 直通(不抑制),第 num_init_frames 個非全零 hop 完成 noise_estimator_init_noise()、anr_is_initialized() 轉 true,但那一個 hop 的 gain 仍是 1,下一個 hop 才開始抑制。數位靜音 hop(所有 bin 恰為 0)不計入 warm-up、也不推進噪聲追蹤(實測 30 個全零 hop 後仍未初始化,gain 維持 1)。初始化之後的全零 hop 不會被強制為 gain=1,照常算 gain,只是不推進噪聲底追蹤器(實測 16k/256、40 個以上噪聲 hop 之後的全零 hop 平均 gain 約 0.2)。
  • MCRA 最小值追蹤窗 L(balanced 實測 64/32/48 幀 ≈ 512 ms)決定噪聲底適應新環境的時間尺度;場景偵測(上半頻帶 γ 超門檻且頻譜平坦度超門檻,連續 scene_change_min_frames 個 hop 才重置)提供快速路徑;broadband_threshold<1 時另有寬頻起音閘。
